cmake_minimum_required(VERSION 3.20)

set(CMAKE_CXX_STANDARD 17)
set(SPIRV_CROSS_CLI OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_GLSL ON CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_CPP OFF CACHE BOOL "" FORCE)
set(SPIRV_CROSS_ENABLE_REFLECT OFF CACHE BOOL "" FORCE)
add_subdirectory(../../lib/rt64/src/contrib/spirv-cross spirv-cross)
add_executable(spirv_cross_msl_ios spirv_cross_msl_ios.cpp)
target_include_directories(spirv_cross_msl_ios PRIVATE ../../lib/rt64/src/contrib/spirv-cross)
target_link_libraries(spirv_cross_msl_ios PRIVATE spirv-cross-core spirv-cross-msl)
